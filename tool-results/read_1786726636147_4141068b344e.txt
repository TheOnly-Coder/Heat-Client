   326→    fun update() {
   327→        if (cancelRun || (noInventoryAttack && (mc.currentScreen is GuiContainer || System.currentTimeMillis() - containerOpen < noInventoryDelay))) return
   328→
   329→        // Update target
   330→        updateTarget()

   333→            mc.gameSettings.thirdPersonView = 1

   342→    /**
   343→     * Tick event
   344→     */
   345→    @EventTarget
   346→    fun onTick(event: TickEvent) {
   347→        if (clickOnly && !mc.gameSettings.keyBindAttack.isKeyDown) return
   348→
   349→        if (blockStatus && autoBlock == "Packet" && releaseAutoBlock && !ignoreTickRule) {
   350→            clicks = 0
   351→            stopBlocking()
   352→            return
   353→        }
   354→
   355→        if (cancelRun) {
   356→            target = null
   357→            hittable = false
   358→            stopBlocking()
   359→            return
   360→        }
   361→
   362→        if (noInventoryAttack && (mc.currentScreen is GuiContainer || System.currentTimeMillis() - containerOpen < noInventoryDelay)) {
   363→            target = null
   364→            hittable = false
   365→            if (mc.currentScreen is GuiContainer) containerOpen = System.currentTimeMillis()
   366→            return
   367→        }
   368→
   369→        if (simulateCooldown && getAttackCooldownProgress() < 1f) {
   370→            return
   371→        }
   372→
   373→        if (target == null && !blockStopInDead) {
   374→            blockStopInDead = true
   375→            stopBlocking()
   376→            return
   377→        }
   378→
   379→        if (target != null) {
   380→            if (mc.thePlayer.getDistanceToEntityBox(target!!) > range && blockStatus) {
   381→                stopBlocking()
   382→                return
   383→            } else if (autoBlock != "Off")
   384→                renderBlocking = true
   385→
   386→            // Usually when you butterfly click, you end up clicking two (and possibly more) times in a single tick.
   387→            // Sometimes you also do not click. The positives outweigh the negatives, however.
   388→            val extraClicks = if (simulateDoubleClicking && !simulateCooldown) nextInt(-1, 1) else 0
   389→
   390→            val maxClicks = clicks + extraClicks
   391→
   392→            repeat(maxClicks) {
   393→                val wasBlocking = blockStatus
   394→
   395→                runAttack(it + 1 == maxClicks)
   396→                clicks--
   397→
   398→                if (wasBlocking && !blockStatus && (releaseAutoBlock && !ignoreTickRule || autoBlock == "Off")) {
   399→                    return
   400→                }
   401→            }
   402→        } else renderBlocking = false
   403→    }

   453→    /**
   454→     * Attack enemy
   455→     */
   456→    @Suppress("VARIABLE_WITH_REDUNDANT_INITIALIZER")
   457→    private fun runAttack(isLastClick: Boolean) {
   458→        var currentTarget = this.target ?: return
   459→
   460→        val thePlayer = mc.thePlayer ?: return
   461→        val theWorld = mc.theWorld ?: return
   462→
   463→        if (noConsumeAttack == "NoHits" && isConsumingItem) {
   464→            return
   465→        }
   466→
   467→        // Settings
   468→        val multi = targetMode == "Multi"
   469→        val manipulateInventory = simulateClosingInventory && !noInventoryAttack && serverOpenInventory
   470→
   471→        // Close inventory when open
   472→        if (manipulateInventory) serverOpenInventory = false
   473→
   474→        updateHittable()
   475→
   476→        currentTarget = this.target ?: return
   477→
   478→        if (hittable && currentTarget.hurtTime > hurtTime) {
   479→            return
   480→        }
   481→
   482→        // Check if enemy is not hittable
   483→        if (!hittable) {
   484→            if (swing != "Off" && failSwing) {
   485→                val rotation = currentRotation ?: thePlayer.rotation
   486→
   487→                // Can humans keep click consistency when performing massive rotation changes?
   488→                // (10-30 rotation difference/doing large mouse movements for example)
   489→                // Maybe apply to attacks too?
   490→                if (getRotationDifference(rotation) > maxRotationDifferenceToSwing) {
   491→                    val lastAttack = attackTickTimes.lastOrNull()?.second ?: 0
   492→
   493→                    // At the same time there is also a chance of the user clicking at least once in a while
   494→                    // when the consistency has dropped a lot.
   495→                    val shouldIgnore = swingWhenTicksLate.isActive() && runTimeTicks - lastAttack >= ticksLateToSwing
   496→
   497→                    if (!shouldIgnore) {
   498→                        return
   499→                    }
   500→                }
   501→
   502→                runWithModifiedRaycastResult(rotation, range, throughWallsRange) {
   503→                    if (swingOnlyInAir && it.typeOfHit != MovingObjectPosition.MovingObjectType.MISS) {
   504→                        return@runWithModifiedRaycastResult
   505→                    }
   506→
   507→                    if (!shouldDelayClick(it.typeOfHit)) {
   508→                        if (it.typeOfHit == MovingObjectPosition.MovingObjectType.ENTITY) {
   509→                            val entity = it.entityHit
   510→
   511→                            // Use own function instead of clickMouse() to maintain keep sprint, auto block, etc
   512→                            if (entity is EntityLivingBase) {
   513→                                attackEntity(entity, isLastClick)
   514→                            }
   515→                        } else {
   516→                            if (swing == "Visual")
   517→                                cancelNextSwing = true
   518→                            // Imitate game click
   519→                            mc.clickMouse()
   520→                            if (swing == "Packet")
   521→                                mc.thePlayer.isSwingInProgress = false
   522→                        }
   523→                        attackTickTimes += it to runTimeTicks
   524→                    }
   525→
   526→                    if (isLastClick) {
   527→                        // We return false because when you click literally once, the attack key's [pressed] status is false.
   528→                        // Since we simulate clicks, we are supposed to respect that behavior.
   529→                        mc.sendClickBlockToController(false)
   530→                    }
   531→                }
   532→            }
   533→        } else {
   534→            blockStopInDead = false
   535→            // Attack
   536→            if (!multi) {
   537→                attackEntity(currentTarget, isLastClick)
   538→            } else {
   539→                var targets = 0
   540→
   541→                for (entity in theWorld.loadedEntityList) {
   542→                    val distance = thePlayer.getDistanceToEntityBox(entity)
   543→
   544→                    if (entity is EntityLivingBase && isEnemy(entity) && distance <= getRange(entity)) {
   545→                        attackEntity(entity, isLastClick)
   546→
   547→                        targets += 1
   548→
   549→                        if (limitedMultiTargets != 0 && limitedMultiTargets <= targets) break
   550→                    }
   551→                }
   552→            }
   553→
   554→            prevTargetEntities += currentTarget.entityId
   555→
   556→            if (target == currentTarget) this.target = null
   557→        }
   558→
   559→        if (targetMode.equals("Switch", ignoreCase = true) && attackTimer.hasTimePassed(switchDelay)) {
   560→            if (switchDelay != 0L) {
   561→                prevTargetEntities += currentTarget.entityId
   562→                attackTimer.reset()
   563→            }
   564→        }
   565→
   566→        // Open inventory
   567→        if (manipulateInventory) serverOpenInventory = true
   568→    }

   570→    /**
   571→     * Update current target
   572→     */
   573→    private fun updateTarget() {
   574→        if (!onScaffold && Scaffold.state)
   575→            return
   576→
   577→        if (!onDestroyBlock && mc.playerController.isHittingBlock)
   578→            return
   579→
   580→        // Reset fixed target to null
   581→        target = null
   582→
   583→        // Settings
   584→        val fov = fov
   585→        val switchMode = targetMode == "Switch"
   586→
   587→        // Find possible targets
   588→        val targets = mutableListOf<EntityLivingBase>()
   589→
   590→        val theWorld = mc.theWorld
   591→        val thePlayer = mc.thePlayer
   592→
   593→        for (entity in theWorld.loadedEntityList) {
   594→            if (entity !is EntityLivingBase || !isEnemy(entity) || (switchMode && entity.entityId in prevTargetEntities)) continue
   595→
   596→            var distance = thePlayer.getDistanceToEntityBox(entity)
   597→
   598→            if (Backtrack.handleEvents()) {
   599→                val trackedDistance = Backtrack.getNearestTrackedDistance(entity)
   600→
   601→                if (distance > trackedDistance) {
   602→                    distance = trackedDistance
   603→                }
   604→            }
   605→
   606→            val entityFov = getRotationDifference(entity)
   607→
   608→            if (distance <= maxRange && (fov == 180F || entityFov <= fov)) {
   609→                if (switchMode && isLookingOnEntities(entity, maxSwitchFOV) || !switchMode) {
   610→                    targets += entity
   611→                }
   612→            }
   613→        }
   614→
   615→        // Sort targets by priority
   616→        when (priority.lowercase()) {
   617→            "distance" -> {
   618→                targets.sortBy {
   619→                    var result = 0.0
   620→
   621→                    Backtrack.runWithNearestTrackedDistance(it) {
   622→                        result = thePlayer.getDistanceToEntityBox(it) // Sort by distance
   623→                    }
   624→
   625→                    result
   626→                }
   627→            }
   628→
   629→            "direction" -> targets.sortBy {
   630→                var result = 0f
   631→
   632→                Backtrack.runWithNearestTrackedDistance(it) {
   633→                    result = getRotationDifference(it) // Sort by FOV
   634→                }
   635→
   636→                result
   637→            }
   638→
   639→            "health" -> targets.sortBy { it.health } // Sort by health
   640→            "livingtime" -> targets.sortBy { -it.ticksExisted } // Sort by existence
   641→            "armor" -> targets.sortBy { it.totalArmorValue } // Sort by armor
   642→            "hurtresistance" -> targets.sortBy { it.hurtResistantTime } // Sort by armor hurt time
   643→            "hurttime" -> targets.sortBy { it.hurtTime } // Sort by hurt time
   644→            "healthabsorption" -> targets.sortBy { it.health + it.absorptionAmount } // Sort by full health with absorption effect
   645→            "regenamplifier" -> targets.sortBy {
   646→                if (it.isPotionActive(Potion.regeneration)) it.getActivePotionEffect(
   647→                    Potion.regeneration
   648→                ).amplifier else -1
   649→            }
   650→        }
   651→
   652→        // Find best target
   653→        for (entity in targets) {
   654→            // Update rotations to current target
   655→            var success = false
   656→
   657→            Backtrack.runWithNearestTrackedDistance(entity) {
   658→                success = updateRotations(entity)
   659→            }
   660→
   661→            if (!success) {
   662→                // when failed then try another target
   663→                continue
   664→            }
   665→
   666→            // Set target to current entity
   667→            target = entity
   668→            return
   669→        }
   670→
   671→        // Cleanup last targets when no target found and try again
   672→        if (prevTargetEntities.isNotEmpty()) {
   673→            prevTargetEntities.clear()
   674→            updateTarget()
   675→        }
   676→    }

   678→    /**
   679→     * Check if [entity] is selected as enemy with current target options and other modules
   680→     */
   681→    private fun isEnemy(entity: Entity?): Boolean {
   682→        if (entity is FakePlayer) {
   683→            return false
   684→        }
   685→
   686→        if (entity is EntityLivingBase && (Dead.state || isAlive(entity)) && entity != mc.thePlayer) {
   687→            if (!Invisible.state && entity.isInvisible) return false
   688→
   689→            if (Players.state && entity is EntityPlayer) {
   690→                if (entity.isSpectator || isBot(entity)) return false
   691→
   692→                if (entity.isClientFriend && !Friends.handleEvents()) return false
   693→
   694→                return !Teams.handleEvents() || !Teams.isInYourTeam(entity)
   695→            }
   696→
   697→            return Mobs.state && entity.isMob || Animals.state && entity.isAnimal
   698→        }
   699→
   700→        return false
   701→    }

   703→    /**
   704→     * Attack [entity]
   705→     */
   706→    private fun attackEntity(entity: EntityLivingBase, isLastClick: Boolean) {
   707→        // Stop blocking
   708→        val thePlayer = mc.thePlayer
   709→
   710→        if (!onScaffold && Scaffold.state)
   711→            return
   712→
   713→        if ((thePlayer.isBlocking || renderBlocking) && (autoBlock == "Off" && blockStatus || autoBlock == "Packet" && releaseAutoBlock)) {
   714→            stopBlocking()
   715→
   716→            if (!ignoreTickRule || autoBlock == "Off") {
   717→                return
   718→            }
   719→        }
   720→
   721→        // The function is only called when we are facing an entity
   722→        if (shouldDelayClick(MovingObjectPosition.MovingObjectType.ENTITY)) {
   723→            return
   724→        }
   725→
   726→        // Call attack event
   727→        callEvent(AttackEvent(entity))
   728→
   729→        // Attack target
   730→        mc.thePlayer.swing(swing)
   731→
   732→        sendPacket(C02PacketUseEntity(entity, ATTACK))
   733→
   734→        if (mc.playerController.currentGameType != WorldSettings.GameType.SPECTATOR)
   735→            thePlayer.attackTargetEntityWithCurrentItem(entity)
   736→
   737→        // FakeSharp
   738→        if (EnchantmentHelper.getModifierForCreature(thePlayer.heldItem, entity.creatureAttribute) <= 0f && fakeSharp)
   739→            thePlayer.onEnchantmentCritical(entity)
   740→
   741→        CPSCounter.registerClick(CPSCounter.MouseButton.LEFT)
   742→
   743→        // Start blocking after attack
   744→        if (autoBlock != "Off" && (thePlayer.isBlocking || canBlock) && isLastClick) {
   745→            startBlocking(entity, interactAutoBlock, autoBlock == "Fake")
   746→        }
   747→
   748→        resetLastAttackedTicks()
   749→    }

   751→    /**
   752→     * Update killaura rotations to enemy
   753→     */
   754→    private fun updateRotations(entity: Entity): Boolean {
   755→        val player = mc.thePlayer ?: return false
   756→
   757→        if (!onScaffold && Scaffold.state)
   758→            return false
   759→
   760→        if (!onDestroyBlock && mc.playerController.isHittingBlock)
   761→            return false
   762→
   763→        val (predictX, predictY, predictZ) = entity.currPos.subtract(entity.prevPos)
   764→            .times(2 + predictEnemyPosition)
   765→
   766→        val boundingBox = entity.hitBox.offset(predictX, predictY, predictZ)
   767→        val (currPos, oldPos) = player.currPos to player.prevPos
   768→
   769→        val simPlayer = SimulatedPlayer.fromClientPlayer(player.movementInput)
   770→
   771→        repeat(predictClientMovement + 1) {
   772→            simPlayer.tick()
   773→        }
   774→
   775→        player.setPosAndPrevPos(simPlayer.pos)
   776→
   777→        val rotation = searchCenter(
   778→            boundingBox,
   779→            outborder && !attackTimer.hasTimePassed(attackDelay / 2),
   780→            randomCenter,
   781→            gaussianOffset = this.gaussianOffset,
   782→            predict = false,
   783→            lookRange = range.toFloat() + scanRange.toFloat(),
   784→            attackRange = range.toFloat(),
   785→            throughWallsRange = throughWallsRange.toFloat()
   786→        )
   787→
   788→        if (rotation == null) {
   789→            player.setPosAndPrevPos(currPos, oldPos)
   790→
   791→            return false
   792→        }
   793→
   794→        // Get our current rotation. Otherwise, player rotation.
   795→        val currentRotation = currentRotation ?: player.rotation
   796→
   797→        var limitedRotation = limitAngleChange(
   798→            currentRotation,
   799→            rotation,
   800→            nextFloat(minTurnSpeed, maxTurnSpeed),
   801→            smootherMode
   802→        )
   803→
   804→        if (micronized) {
   805→            // Is player facing the entity with current rotation?
   806→            if (isRotationFaced(entity, maxRange, currentRotation)) {
   807→                // Limit angle change but this time modify the turn speed.
   808→                limitedRotation = limitAngleChange(
   809→                    currentRotation, rotation,
   810→                    nextFloat(endInclusive = micronizedStrength)
   811→                )
   812→            }
   813→        }
   814→
   815→        if (silentRotation) {
   816→            setTargetRotation(
   817→                limitedRotation,
   818→                keepRotationTicks,
   819→                !(!silentRotation || rotationStrafe == "Off"),
   820→                rotationStrafe == "Strict",
   821→                minTurnSpeed to maxTurnSpeed,
   822→                angleThresholdUntilReset,
   823→                smootherMode
   824→            )
   825→        } else {
   826→            limitedRotation.toPlayer(player)
   827→        }
   828→
   829→        player.setPosAndPrevPos(currPos, oldPos)
   830→
   831→        return true
   832→    }

   834→    /**
   835→     * Check if enemy is hittable with current rotations
   836→     */
   837→    private fun updateHittable() {
   838→        val eyes = mc.thePlayer.eyes
   839→
   840→        val currentRotation = currentRotation ?: mc.thePlayer.rotation
   841→        val target = this.target ?: return
   842→
   843→        if (!onScaffold && Scaffold.state)
   844→            return
   845→
   846→        if (!onDestroyBlock && mc.playerController.isHittingBlock)
   847→            return
   848→
   849→        var chosenEntity: Entity? = null
   850→
   851→        if (raycast) {
   852→            chosenEntity = raycastEntity(
   853→                range,
   854→                currentRotation.yaw,
   855→                currentRotation.pitch
   856→            ) { entity -> !livingRaycast || entity is EntityLivingBase && entity !is EntityArmorStand }
   857→
   858→            if (chosenEntity != null && chosenEntity is EntityLivingBase && (Friends.handleEvents() || !(chosenEntity is EntityPlayer && chosenEntity.isClientFriend))) {
   859→                if (raycastIgnored && target != chosenEntity) {
   860→                    this.target = chosenEntity
   861→                }
   862→            }
   863→
   864→            hittable = this.target == chosenEntity
   865→        } else {
   866→            hittable = isRotationFaced(target, range, currentRotation)
   867→        }
   868→
   869→        if (!hittable) {
   870→            return
   871→        }
   872→
   873→        val targetToCheck = chosenEntity ?: this.target ?: return
   874→
   875→        // If player is inside entity, automatic yes because the intercept below cannot check for that
   876→        // Minecraft does the same, see #EntityRenderer line 353
   877→        if (targetToCheck.hitBox.isVecInside(eyes)) {
   878→            return
   879→        }
   880→
   881→        var checkNormally = true
   882→
   883→        if (Backtrack.handleEvents()) {
   884→            Backtrack.loopThroughBacktrackData(targetToCheck) {
   885→                if (targetToCheck.hitBox.isVecInside(eyes)) {
   886→                    checkNormally = false
   887→                    return@loopThroughBacktrackData true
   888→                }
   889→
   890→                // Recreate raycast logic
   891→                val intercept = targetToCheck.hitBox.calculateIntercept(
   892→                    eyes,
   893→                    eyes + getVectorForRotation(currentRotation) * range
   894→                )
   895→
   896→                if (intercept != null) {
   897→                    // Is the entity box raycast vector visible? If not, check through-wall range
   898→                    hittable =
   899→                        isVisible(intercept.hitVec) || mc.thePlayer.getDistanceToEntityBox(targetToCheck) <= throughWallsRange
   900→
   901→                    if (hittable) {
   902→                        checkNormally = false
   903→                        return@loopThroughBacktrackData true
   904→                    }
   905→                }
   906→
   907→                return@loopThroughBacktrackData false
   908→            }
   909→        }
   910→
   911→        if (!checkNormally) {
   912→            return
   913→        }
   914→
   915→        // Recreate raycast logic
   916→        val intercept = targetToCheck.hitBox.calculateIntercept(
   917→            eyes,
   918→            eyes + getVectorForRotation(currentRotation) * range
   919→        )
   920→
   921→        // Is the entity box raycast vector visible? If not, check through-wall range
   922→        hittable =
   923→            isVisible(intercept.hitVec) || mc.thePlayer.getDistanceToEntityBox(targetToCheck) <= throughWallsRange
   924→    }

   929→    private fun startBlocking(interactEntity: Entity, interact: Boolean, fake: Boolean = false) {
   930→        if (blockStatus && !uncpAutoBlock)
   931→            return
   932→
   933→        if (!onScaffold && Scaffold.state)
   934→            return
   935→
   936→        if (!onDestroyBlock && mc.playerController.isHittingBlock)
   937→            return
   938→

   940→            blockStatus = true
   941→            renderBlocking = true
   942→            return
   943→        }
   944→
   945→        if (!fake) {
   946→            if (!(blockRate > 0 && nextInt(endExclusive = 100) <= blockRate)) return
   947→

   949→                val positionEye = mc.thePlayer.eyes
   950→
   951→                val boundingBox = interactEntity.hitBox
   952→
   953→                val (yaw, pitch) = currentRotation ?: mc.thePlayer.rotation
   954→
   955→                val vec = getVectorForRotation(Rotation(yaw, pitch))
   956→
   957→                val lookAt = positionEye.add(vec * maxRange)
   958→
   959→                val movingObject = boundingBox.calculateIntercept(positionEye, lookAt) ?: return
   960→                val hitVec = movingObject.hitVec
   961→
   962→                sendPackets(
   963→                    C02PacketUseEntity(interactEntity, hitVec - interactEntity.positionVector),
   964→                    C02PacketUseEntity(interactEntity, INTERACT)
   965→                )
   966→
   967→            }
   968→
   969→            if (switchStartBlock) {
   970→                InventoryUtils.serverSlot = (InventoryUtils.serverSlot + 1) % 9
   971→                InventoryUtils.serverSlot = mc.thePlayer.inventory.currentItem
   972→            }
   973→
   974→            sendPacket(C08PacketPlayerBlockPlacement(mc.thePlayer.heldItem))
   975→            blockStatus = true
   976→        }
   977→
   978→        renderBlocking = true
   979→
   980→        CPSCounter.registerClick(CPSCounter.MouseButton.RIGHT)
   981→    }
