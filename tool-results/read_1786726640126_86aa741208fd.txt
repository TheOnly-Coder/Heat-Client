   343→   343→     * Tick event
   344→   344→     */
   345→   345→    @EventTarget
   346→   346→    fun onTick(event: TickEvent) {
   347→   347→        if (clickOnly && !mc.gameSettings.keyBindAttack.isKeyDown) return
   348→   348→
   349→   349→        if (blockStatus && autoBlock == "Packet" && releaseAutoBlock && !ignoreTickRule) {
   350→   350→            clicks = 0
   351→   351→            stopBlocking()
   352→   352→            return
   353→   353→        }
   354→   354→
   355→   355→        if (cancelRun) {
   356→   356→            target = null
   357→   357→            hittable = false
   358→   358→            stopBlocking()
   359→   359→            return
   360→   360→        }
   361→   361→
   362→   362→        if (noInventoryAttack && (mc.currentScreen is GuiContainer || System.currentTimeMillis() - containerOpen < noInventoryDelay)) {
   363→   363→            target = null
   364→   364→            hittable = false
   365→   365→            if (mc.currentScreen is GuiContainer) containerOpen = System.currentTimeMillis()
   366→   366→            return
   367→   367→        }
   368→   368→
   369→   369→        if (simulateCooldown && getAttackCooldownProgress() < 1f) {
   370→   370→            return
   371→   371→        }
   372→   372→
   373→   373→        if (target == null && !blockStopInDead) {
   374→   374→            blockStopInDead = true
   375→   375→            stopBlocking()
   376→   376→            return
   377→   377→        }
   378→   378→
   379→   379→        if (target != null) {
   380→   380→            if (mc.thePlayer.getDistanceToEntityBox(target!!) > range && blockStatus) {
   381→   381→                stopBlocking()
   382→   382→                return
   383→   383→            } else if (autoBlock != "Off")
   384→   384→                renderBlocking = true
   385→   385→
   386→   386→            // Usually when you butterfly click, you end up clicking two (and possibly more) times in a single tick.
   387→   387→            // Sometimes you also do not click. The positives outweigh the negatives, however.
   388→   388→            val extraClicks = if (simulateDoubleClicking && !simulateCooldown) nextInt(-1, 1) else 0
   389→   389→
   390→   390→            val maxClicks = clicks + extraClicks
   391→   391→
   392→   392→            repeat(maxClicks) {
   393→   393→                val wasBlocking = blockStatus
   394→   394→
   395→   395→                runAttack(it + 1 == maxClicks)
   396→   396→                clicks--
   397→   397→
   398→   398→                if (wasBlocking && !blockStatus && (releaseAutoBlock && !ignoreTickRule || autoBlock == "Off")) {
   399→   399→                    return
   400→   400→                }
   401→   401→            }
   402→   402→        } else renderBlocking = false
   403→   403→    }

   454→   454→     * Attack enemy
   455→   455→     */
   456→   456→    @Suppress("VARIABLE_WITH_REDUNDANT_INITIALIZER")
   457→   457→    private fun runAttack(isLastClick: Boolean) {
   458→   458→        var currentTarget = this.target ?: return
   459→   459→
   460→   460→        val thePlayer = mc.thePlayer ?: return
   461→   461→        val theWorld = mc.theWorld ?: return
   462→   462→
   463→   463→        if (noConsumeAttack == "NoHits" && isConsumingItem) {
   464→   464→            return
   465→   465→        }
   466→   466→
   467→   467→        // Settings
   468→   468→        val multi = targetMode == "Multi"
   469→   469→        val manipulateInventory = simulateClosingInventory && !noInventoryAttack && serverOpenInventory
   470→   470→
   471→   471→        // Close inventory when open
   472→   472→        if (manipulateInventory) serverOpenInventory = false
   473→   473→
   474→   474→        updateHittable()
   475→   475→
   476→   476→        currentTarget = this.target ?: return
   477→   477→
   478→   478→        if (hittable && currentTarget.hurtTime > hurtTime) {
   479→   479→            return
   480→   480→        }
   481→   481→
   482→   482→        // Check if enemy is not hittable
   483→   483→        if (!hittable) {
   484→   484→            if (swing != "Off" && failSwing) {
   485→   485→                val rotation = currentRotation ?: thePlayer.rotation
   486→   486→
   487→   487→                // Can humans keep click consistency when performing massive rotation changes?
   488→   488→                // (10-30 rotation difference/doing large mouse movements for example)
   489→   489→                // Maybe apply to attacks too?
   490→   490→                if (getRotationDifference(rotation) > maxRotationDifferenceToSwing) {
   491→   491→                    val lastAttack = attackTickTimes.lastOrNull()?.second ?: 0
   492→   492→
   493→   493→                    // At the same time there is also a chance of the user clicking at least once in a while
   494→   494→                    // when the consistency has dropped a lot.
   495→   495→                    val shouldIgnore = swingWhenTicksLate.isActive() && runTimeTicks - lastAttack >= ticksLateToSwing
   496→   496→
   497→   497→                    if (!shouldIgnore) {
   498→   498→                        return
   499→   499→                    }
   500→   500→                }
   501→   501→
   502→   502→                runWithModifiedRaycastResult(rotation, range, throughWallsRange) {
   503→   503→                    if (swingOnlyInAir && it.typeOfHit != MovingObjectPosition.MovingObjectType.MISS) {
   504→   504→                        return@runWithModifiedRaycastResult
   505→   505→                    }
   506→   506→
   507→   507→                    if (!shouldDelayClick(it.typeOfHit)) {
   508→   508→                        if (it.typeOfHit == MovingObjectPosition.MovingObjectType.ENTITY) {
   509→   509→                            val entity = it.entityHit
   510→   510→
   511→   511→                            // Use own function instead of clickMouse() to maintain keep sprint, auto block, etc
   512→   512→                            if (entity is EntityLivingBase) {
   513→   513→                                attackEntity(entity, isLastClick)
   514→   514→                            }
   515→   515→                        } else {
   516→   516→                            if (swing == "Visual")
   517→   517→                                cancelNextSwing = true
   518→   518→                            // Imitate game click
   519→   519→                            mc.clickMouse()
   520→   520→                            if (swing == "Packet")
   521→   521→                                mc.thePlayer.isSwingInProgress = false
   522→   522→                        }
   523→   523→                        attackTickTimes += it to runTimeTicks
   524→   524→                    }
   525→   525→
   526→   526→                    if (isLastClick) {
   527→   527→                        // We return false because when you click literally once, the attack key's [pressed] status is false.
   528→   528→                        // Since we simulate clicks, we are supposed to respect that behavior.
   529→   529→                        mc.sendClickBlockToController(false)
   530→   530→                    }
   531→   531→                }
   532→   532→            }
   533→   533→        } else {
   534→   534→            blockStopInDead = false
   535→   535→            // Attack
   536→   536→            if (!multi) {
   537→   537→                attackEntity(currentTarget, isLastClick)
   538→   538→            } else {
   539→   539→                var targets = 0
   540→   540→
   541→   541→                for (entity in theWorld.loadedEntityList) {
   542→   542→                    val distance = thePlayer.getDistanceToEntityBox(entity)
   543→   543→
   544→   544→                    if (entity is EntityLivingBase && isEnemy(entity) && distance <= getRange(entity)) {
   545→   545→                        attackEntity(entity, isLastClick)
   546→   546→
   547→   547→                        targets += 1
   548→   548→
   549→   549→                        if (limitedMultiTargets != 0 && limitedMultiTargets <= targets) break
   550→   550→                    }
   551→   551→                }
   552→   552→            }
   553→   553→
   554→   554→            prevTargetEntities += currentTarget.entityId
   555→   555→
   556→   556→            if (target == currentTarget) this.target = null
   557→   557→        }
   558→   558→
   559→   559→        if (targetMode.equals("Switch", ignoreCase = true) && attackTimer.hasTimePassed(switchDelay)) {
   560→   560→            if (switchDelay != 0L) {
   561→   561→                prevTargetEntities += currentTarget.entityId
   562→   562→                attackTimer.reset()
   563→   563→            }
   564→   564→        }
   565→   565→
   566→   566→        // Open inventory
   567→   567→        if (manipulateInventory) serverOpenInventory = true
   568→   568→    }

   570→   570→    /**
   571→   571→     * Update current target
   572→   572→     */
   573→   573→    private fun updateTarget() {
   574→   574→        if (!onScaffold && Scaffold.state)
   575→   575→            return
   576→   576→
   577→   577→        if (!onDestroyBlock && mc.playerController.isHittingBlock)
   578→   578→            return
   579→   579→
   580→   580→        // Reset fixed target to null
   581→   581→        target = null
   582→   582→
   583→   583→        // Settings
   584→   584→        val fov = fov
   585→   585→        val switchMode = targetMode == "Switch"
   586→   586→
   587→   587→        // Find possible targets
   588→   588→        val targets = mutableListOf<EntityLivingBase>()
   589→   589→
   590→   590→        val theWorld = mc.theWorld
   591→   591→        val thePlayer = mc.thePlayer
   592→   592→
   593→   593→        for (entity in theWorld.loadedEntityList) {
   594→   594→            if (entity !is EntityLivingBase || !isEnemy(entity) || (switchMode && entity.entityId in prevTargetEntities)) continue
   595→   595→
   596→   596→            var distance = thePlayer.getDistanceToEntityBox(entity)
   597→   597→
   598→   598→            if (Backtrack.handleEvents()) {
   599→   599→                val trackedDistance = Backtrack.getNearestTrackedDistance(entity)
   600→   600→
   601→   601→                if (distance > trackedDistance) {
   602→   602→                    distance = trackedDistance
   603→   603→                }
   604→   604→            }
   605→   605→
   606→   606→            val entityFov = getRotationDifference(entity)
   607→   607→
   608→   608→            if (distance <= maxRange && (fov == 180F || entityFov <= fov)) {
   609→   609→                if (switchMode && isLookingOnEntities(entity, maxSwitchFOV) || !switchMode) {
   610→   610→                    targets += entity
   611→   611→                }
   612→   612→            }
   613→   613→        }
   614→   614→
   615→   615→        // Sort targets by priority
   616→   616→        when (priority.lowercase()) {
   617→   617→            "distance" -> {
   618→   618→                targets.sortBy {
   619→   619→                    var result = 0.0
   620→   620→
   621→   621→                    Backtrack.runWithNearestTrackedDistance(it) {
   622→   622→                        result = thePlayer.getDistanceToEntityBox(it) // Sort by distance
   623→   623→                    }
   624→   624→
   625→   625→                    result
   626→   626→                }
   627→   627→            }
   628→   628→
   629→   629→            "direction" -> targets.sortBy {
   630→   630→                var result = 0f
   631→   631→
   632→   632→                Backtrack.runWithNearestTrackedDistance(it) {
   633→   633→                    result = getRotationDifference(it) // Sort by FOV
   634→   634→                }
   635→   635→
   636→   636→                result
   637→   637→            }
   638→   638→
   639→   639→            "health" -> targets.sortBy { it.health } // Sort by health
   640→   640→            "livingtime" -> targets.sortBy { -it.ticksExisted } // Sort by existence
   641→   641→            "armor" -> targets.sortBy { it.totalArmorValue } // Sort by armor
   642→   642→            "hurtresistance" -> targets.sortBy { it.hurtResistantTime } // Sort by armor hurt time
   643→   643→            "hurttime" -> targets.sortBy { it.hurtTime } // Sort by hurt time
   644→   644→            "healthabsorption" -> targets.sortBy { it.health + it.absorptionAmount } // Sort by full health with absorption effect
   645→   645→            "regenamplifier" -> targets.sortBy {
   646→   646→                if (it.isPotionActive(Potion.regeneration)) it.getActivePotionEffect(
   647→   647→                    Potion.regeneration
   648→   648→                ).amplifier else -1
   649→   649→            }
   650→   650→        }
   651→   651→
   652→   652→        // Find best target
   653→   653→        for (entity in targets) {
   654→   654→            // Update rotations to current target
   655→   655→            var success = false
   656→   656→
   657→   657→            Backtrack.runWithNearestTrackedDistance(entity) {
   658→   658→                success = updateRotations(entity)
   659→   659→            }
   660→   660→
   661→   661→            if (!success) {
   662→   662→                // when failed then try another target
   663→   663→                continue
   664→   664→            }
   665→   665→
   666→   666→            // Set target to current entity
   667→   667→            target = entity
   668→   668→            return
   669→   669→        }
   670→   670→
   671→   671→        // Cleanup last targets when no target found and try again
   672→   672→        if (prevTargetEntities.isNotEmpty()) {
   673→   673→            prevTargetEntities.clear()
   674→   674→            updateTarget()
   675→   675→        }
   676→   676→    }

   679→   679→     * Check if [entity] is selected as enemy with current target options and other modules
   680→   680→     */
   681→   681→    private fun isEnemy(entity: Entity?): Boolean {
   682→   682→        if (entity is FakePlayer) {
   683→   683→            return false
   684→   684→        }
   685→   685→
   686→   686→        if (entity is EntityLivingBase && (Dead.state || isAlive(entity)) && entity != mc.thePlayer) {
   687→   687→            if (!Invisible.state && entity.isInvisible) return false
   688→   688→
   689→   689→            if (Players.state && entity is EntityPlayer) {
   690→   690→                if (entity.isSpectator || isBot(entity)) return false
   691→   691→
   692→   692→                if (entity.isClientFriend && !Friends.handleEvents()) return false
   693→   693→
   694→   694→                return !Teams.handleEvents() || !Teams.isInYourTeam(entity)
   695→   695→            }
   696→   696→
   697→   697→            return Mobs.state && entity.isMob || Animals.state && entity.isAnimal
   698→   698→        }
   699→   699→
   700→   700→        return false
   701→   701→    }

   754→   754→    private fun updateRotations(entity: Entity): Boolean {
   755→   755→        val player = mc.thePlayer ?: return false
   756→   756→
   757→   757→        if (!onScaffold && Scaffold.state)
   758→   758→            return false
   759→   759→
   760→   760→        if (!onDestroyBlock && mc.playerController.isHittingBlock)
   761→   761→            return false
   762→   762→
   763→   763→        val (predictX, predictY, predictZ) = entity.currPos.subtract(entity.prevPos)
   764→   764→            .times(2 + predictEnemyPosition)
   765→   765→
   766→   766→        val boundingBox = entity.hitBox.offset(predictX, predictY, predictZ)
   767→   767→        val (currPos, oldPos) = player.currPos to player.prevPos
   768→   768→
   769→   769→        val simPlayer = SimulatedPlayer.fromClientPlayer(player.movementInput)
   770→   770→
   771→   771→        repeat(predictClientMovement + 1) {
   772→   772→            simPlayer.tick()
   773→   773→        }
   774→   774→
   775→   775→        player.setPosAndPrevPos(simPlayer.pos)
   776→   776→
   777→   777→        val rotation = searchCenter(
   778→   778→            boundingBox,
   779→   779→            outborder && !attackTimer.hasTimePassed(attackDelay / 2),
   780→   780→            randomCenter,
   781→   781→            gaussianOffset = this.gaussianOffset,
   782→   782→            predict = false,
   783→   783→            lookRange = range.toFloat() + scanRange.toFloat(),
   784→   784→            attackRange = range.toFloat(),
   785→   785→            throughWallsRange = throughWallsRange.toFloat()
   786→   786→        )
   787→   787→
   788→   788→        if (rotation == null) {
   789→   789→            player.setPosAndPrevPos(currPos, oldPos)
   790→   790→
   791→   791→            return false
   792→   792→        }
   793→   793→
   794→   794→        // Get our current rotation. Otherwise, player rotation.
   795→   795→        val currentRotation = currentRotation ?: player.rotation
   796→   796→
   797→   797→        var limitedRotation = limitAngleChange(
   798→   798→            currentRotation,
   799→   799→            rotation,
   800→   800→            nextFloat(minTurnSpeed, maxTurnSpeed),
   801→   801→            smootherMode
   802→   802→        )
   803→   803→
   804→   804→        if (micronized) {
   805→   805→            // Is player facing the entity with current rotation?
   806→   806→            if (isRotationFaced(entity, maxRange, currentRotation)) {
   807→   807→                // Limit angle change but this time modify the turn speed.
   808→   808→                limitedRotation = limitAngleChange(
   809→   809→                    currentRotation, rotation,
   810→   810→                    nextFloat(endInclusive = micronizedStrength)
   811→   811→                )
   812→   812→            }
   813→   813→        }
   814→   814→
   815→   815→        if (silentRotation) {
   816→   816→            setTargetRotation(
   817→   817→                limitedRotation,
   818→   818→                keepRotationTicks,
   819→   819→                !(!silentRotation || rotationStrafe == "Off"),
   820→   820→                rotationStrafe == "Strict",
   821→   821→                minTurnSpeed to maxTurnSpeed,
   822→   822→                angleThresholdUntilReset,
   823→   823→                smootherMode
   824→   824→            )
   825→   825→        } else {
   826→   826→            limitedRotation.toPlayer(player)
   827→   827→        }
   828→   828→
   829→   829→        player.setPosAndPrevPos(currPos, oldPos)
   830→   830→
   831→   831→        return true
   832→   832→    }

   834→   834→    /**
   835→   835→     * Check if enemy is hittable with current rotations
   836→   836→     */
   837→   837→    private fun updateHittable() {
   838→   838→        val eyes = mc.thePlayer.eyes
   839→   839→
   840→   840→        val currentRotation = currentRotation ?: mc.thePlayer.rotation
   841→   841→        val target = this.target ?: return
   842→   842→
   843→   843→        if (!onScaffold && Scaffold.state)
   844→   844→            return
   845→   845→
   846→   846→        if (!onDestroyBlock && mc.playerController.isHittingBlock)
   847→   847→            return
   848→   848→
   849→   849→        var chosenEntity: Entity? = null
   850→   850→
   851→   851→        if (raycast) {
   852→   852→            chosenEntity = raycastEntity(
   853→   853→                range,
   854→   854→                currentRotation.yaw,
   855→   855→                currentRotation.pitch
   856→   856→            ) { entity -> !livingRaycast || entity is EntityLivingBase && entity !is EntityArmorStand }
   857→   857→
   858→   858→            if (chosenEntity != null && chosenEntity is EntityLivingBase && (Friends.handleEvents() || !(chosenEntity is EntityPlayer && chosenEntity.isClientFriend))) {
   859→   859→                if (raycastIgnored && target != chosenEntity) {
   860→   860→                    this.target = chosenEntity
   861→   861→                }
   862→   862→            }
   863→   863→
   864→   864→            hittable = this.target == chosenEntity
   865→   865→        } else {
   866→   866→            hittable = isRotationFaced(target, range, currentRotation)
   867→   867→        }
   868→   868→
   869→   869→        if (!hittable) {
   870→   870→            return
   871→   871→        }
   872→   872→
   873→   873→        val targetToCheck = chosenEntity ?: this.target ?: return
   874→   874→
   875→   875→        // If player is inside entity, automatic yes because the intercept below cannot check for that
   876→   876→        // Minecraft does the same, see #EntityRenderer line 353
   877→   877→        if (targetToCheck.hitBox.isVecInside(eyes)) {
   878→   878→            return
   879→   879→        }
   880→   880→
   881→   881→        var checkNormally = true
   882→   882→
   883→   883→        if (Backtrack.handleEvents()) {
   884→   884→            Backtrack.loopThroughBacktrackData(targetToCheck) {
   885→   885→                if (targetToCheck.hitBox.isVecInside(eyes)) {
   886→   886→                    checkNormally = false
   887→   887→                    return@loopThroughBacktrackData true
   888→   888→                }
   889→   889→
   890→   890→                // Recreate raycast logic
   891→   891→                val intercept = targetToCheck.hitBox.calculateIntercept(
   892→   892→                    eyes,
   893→   893→                    eyes + getVectorForRotation(currentRotation) * range
   894→   894→                )
   895→   895→
   896→   896→                if (intercept != null) {
   897→   897→                    // Is the entity box raycast vector visible? If not, check through-wall range
   898→   898→                    hittable =
   899→   899→                        isVisible(intercept.hitVec) || mc.thePlayer.getDistanceToEntityBox(targetToCheck) <= throughWallsRange
   900→   900→
   901→   901→                    if (hittable) {
   902→   902→                        checkNormally = false
   903→   903→                        return@loopThroughBacktrackData true
   904→   904→                    }
   905→   905→                }
   906→   906→
   907→   907→                return@loopThroughBacktrackData false
   908→   908→            }
   909→   909→        }
   910→   910→
   911→   911→        if (!checkNormally) {
   912→   912→            return
   913→   913→        }
   914→   914→
   915→   915→        // Recreate raycast logic
   916→   916→        val intercept = targetToCheck.hitBox.calculateIntercept(
   917→   917→            eyes,
   918→   918→            eyes + getVectorForRotation(currentRotation) * range
   919→   919→        )
   920→   920→
   921→   921→        // Is the entity box raycast vector visible? If not, check through-wall range
   922→   922→        hittable =
   923→   923→            isVisible(intercept.hitVec) || mc.thePlayer.getDistanceToEntityBox(targetToCheck) <= throughWallsRange
   924→   924→    }

   926→   926→    /**
   927→   927→     * Start blocking
   928→   928→     */
   929→   929→    private fun startBlocking(interactEntity: Entity, interact: Boolean, fake: Boolean = false) {
   930→   930→        if (blockStatus && !uncpAutoBlock)
   931→   931→            return
   932→   932→
   933→   933→        if (!onScaffold && Scaffold.state)
   934→   934→            return
   935→   935→
   936→   936→        if (!onDestroyBlock && mc.playerController.isHittingBlock)
   937→   937→            return
   938→   938→
   939→   939→        if (mc.thePlayer.isBlocking) {
   940→   940→            blockStatus = true
   941→   941→            renderBlocking = true
   942→   942→            return
   943→   943→        }
   944→   944→
   945→   945→        if (!fake) {
   946→   946→            if (!(blockRate > 0 && nextInt(endExclusive = 100) <= blockRate)) return
   947→   947→
   948→   948→            if (interact) {
   949→   949→                val positionEye = mc.thePlayer.eyes
   950→   950→
   951→   951→                val boundingBox = interactEntity.hitBox
   952→   952→
   953→   953→                val (yaw, pitch) = currentRotation ?: mc.thePlayer.rotation
   954→   954→
