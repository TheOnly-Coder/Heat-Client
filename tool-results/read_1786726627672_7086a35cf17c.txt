   160→    fun updateTagDetails() {
   161→        val pair: Pair<String, String> = when (tagsStyle) {
   162→            "[]", "()", "<>" -> tagsStyle[0].toString() to tagsStyle[1].toString()
   163→            "-", "|" -> tagsStyle[0] + " " to ""
   164→            else -> "" to ""
   165→        }
   166→
   167→        tagPrefix = (if (tagsArrayColor) " " else " §7") + pair.first
   168→        tagSuffix = pair.second
   169→    }

   171→    private fun getDisplayString(module: Module): String {
   172→        val moduleName = when (moduleCase) {
   173→            "Uppercase" -> module.getName().uppercase()
   174→            "Lowercase" -> module.getName().lowercase()
   175→            else -> module.getName()
   176→        }
   177→
   178→        var tag = module.tag ?: ""
   179→
   180→        tag = when (tagsCase) {
   181→            "Uppercase" -> tag.uppercase()
   182→            "Lowercase" -> tag.lowercase()
   183→            else -> tag
   184→        }
   185→
   186→        val moduleTag = if (tags && !module.tag.isNullOrEmpty()) tagPrefix + tag + tagSuffix else ""
   187→
   188→        return moduleName + moduleTag
   189→    }

   215→                        }
   216→                    } else {

   218→                    }
